    "max_loss": float | None,
    "breakeven": float | None,
    "profit_target": float | None,
    "hard_stop": float | None,

    # P&L scenarios
    "scenarios": list | None,           # list of ScenarioPnL dicts
    "probability_weighted_ev": float | None,
    "slippage_adj_ev": float | None,

    # Kelly sizing
    "kelly_dollars": float | None,
    "kelly_contracts": int | None,

    "paragraph_2": str | None,
    "paragraph_3": str | None,

**Implementation — chain in order:**

```python
import logging
import dataclasses
from datetime import date
from typing import Optional

logger = logging.getLogger(__name__)

CRYPTO_ETF_ASSET_CLASSES = {"crypto_etf"}

def run_recommendation(
    ticker: str,
    analytics_result: dict,
    fundamentals_result: dict,
    ticker_info,
    chain: dict,
    portfolio_value: float = 100_000.0,
) -> dict:
    """Orchestrate the full recommendation pipeline for a single Stage 2 candidate.

    Returns a complete result dict (never raises). On any unhandled exception,
    returns {"ticker": ticker, "error": str(exc), "passed": False}.
    """
    ticker = ticker.upper()
    base_result = {"ticker": ticker, "error": None, "passed": False}

    try:
        from recommendations.structures import select_structure, select_strikes
        from recommendations.pnl import compute_pnl_scenarios, compute_kelly_size, PnLResult
        from recommendations.gonogo import evaluate_gonogo
        from recommendations.card import build_recommendation_card

        today = date.today()
        spot = chain.get("underlying_price", 0.0)

        # Step 1: Pre-select structure (needed for go/no-go HARD-08 and HARD-09)
        proposed_structure = select_structure(ticker_info, analytics_result)

        # Step 2: Select strikes and DTE (needed for slippage EV for HARD-03)
        structure_result = select_strikes(analytics_result, ticker_info, chain, today)

        # Step 3: Compute P&L to get slippage_adj_ev (needed by go/no-go HARD-03)
        pnl_scenarios_out = compute_pnl_scenarios(structure_result, chain, analytics_result)
        (scenarios, net_credit, max_loss, breakeven,
         profit_target, hard_stop, roll_trigger) = pnl_scenarios_out

        slippage_adj_ev = net_credit  # net credit per contract (in dollars)

        # Step 4: Go/No-Go evaluation
        gonogo_result = evaluate_gonogo(
            analytics_result=analytics_result,
            fundamentals_result=fundamentals_result,
            ticker_info=ticker_info,
            proposed_structure=proposed_structure,
            slippage_adj_ev=slippage_adj_ev,
        )

        # Step 5: Kelly sizing
        from dataclasses import asdict as dc_asdict

        # Build PnLResult for Kelly computation
        import numpy as np
        pnl_result = PnLResult(
            net_credit=net_credit,
            max_loss=max_loss,
            breakeven=breakeven,
            profit_target=profit_target,
            hard_stop=hard_stop,
            roll_trigger=roll_trigger,
            scenarios=scenarios,
            probability_weighted_ev=float(np.sum([s.weighted_pnl for s in scenarios])),
            slippage_adj_ev=slippage_adj_ev,
            kelly_dollars=0.0,
            kelly_contracts=0,
            kelly_pct=0.0,
            kelly_breakdown={},
        )
        kelly_dollars, kelly_contracts, kelly_pct, kelly_breakdown = compute_kelly_size(
            analytics_result, ticker_info, pnl_result, portfolio_value, spot=spot
        )
        # Update pnl_result with Kelly values
        pnl_result = dataclasses.replace(
            pnl_result,
            kelly_dollars=kelly_dollars,
            kelly_contracts=kelly_contracts,
            kelly_pct=kelly_pct,
            kelly_breakdown=kelly_breakdown,
        )

        # Step 6: Build recommendation card (always, even if gonogo failed — for UI)
        card = build_recommendation_card(
            ticker=ticker,
            analytics_result=analytics_result,
            fundamentals_result=fundamentals_result,
            ticker_info=ticker_info,
            chain=chain,
            gonogo_result=gonogo_result,
            structure_result=structure_result,
            pnl_result=pnl_result,
        )

        # Convert ScenarioPnL list to plain dicts for JSON serialization
        scenarios_dicts = [dc_asdict(s) for s in scenarios]

        result = {
            "ticker": ticker,
            "error": None,
            "passed": gonogo_result.passed,
            "gonogo_summary": card.gonogo_summary,
            "gonogo_checks": gonogo_result.checks,
            "structure": card.structure,
            "expiration_date": card.expiration_date,
            "dte": card.dte,
            "short_strike": card.short_strike,
            "long_strike": card.long_strike,
            "spot": card.spot,
            "target_delta": card.target_delta,
            "fomc_status": card.fomc_status,
            "fomc_message": card.fomc_message,
            "net_credit": card.net_credit,
            "max_loss": card.max_loss,
            "breakeven": card.breakeven,
            "profit_target": card.profit_target,
            "hard_stop": card.hard_stop,
            "roll_trigger": card.roll_trigger,
            "scenarios": scenarios_dicts,
            "probability_weighted_ev": card.probability_weighted_ev,
            "slippage_adj_ev": card.slippage_adj_ev,
            "kelly_dollars": card.kelly_dollars,
            "kelly_contracts": card.kelly_contracts,
            "kelly_pct": card.kelly_pct,
            "kelly_breakdown": card.kelly_breakdown,
            "paragraph_1": card.paragraph_1,
            "paragraph_2": card.paragraph_2,
            "paragraph_3": card.paragraph_3,
            "order_text": card.order_text,
            "card": dc_asdict(card),
        }

        logger.info(
            "[REC] %s passed=%s structure=%s strike=%.1f dte=%d ev=%.2f kelly=%d contracts",
            ticker, gonogo_result.passed, card.structure,
            card.short_strike or 0, card.dte or 0,
            card.slippage_adj_ev or 0, card.kelly_contracts or 0,
        )
        return result

    except Exception as exc:
        logger.error("[REC] run_recommendation(%s) failed: %s", ticker, exc, exc_info=True)
        return {**base_result, "error": str(exc), "passed": False,
                **{k: None for k in ["gonogo_summary","gonogo_checks","structure","expiration_date",
                                      "dte","short_strike","long_strike","spot","target_delta",
                                      "fomc_status","fomc_message","net_credit","max_loss","breakeven",
                                      "profit_target","hard_stop","roll_trigger","scenarios",
                                      "probability_weighted_ev","slippage_adj_ev","kelly_dollars",
                                      "kelly_contracts","kelly_pct","kelly_breakdown",
                                      "paragraph_1","paragraph_2","paragraph_3","order_text","card"]}}
```
