**normalize_signal(value: float, min_val: float, max_val: float) -> float**

Clamps value to [min_val, max_val] and linearly scales to [0, 1]:
```python
def normalize_signal(value, min_val, max_val):
    if max_val <= min_val:
        return 0.0
    return float(np.clip((value - min_val) / (max_val - min_val), 0.0, 1.0))
```

**composite_vrp_score(signals: dict, has_earnings_in_window: bool = False, near_term_earnings: bool = False) -> float**

Exact 12-signal weighted sum from PRD §6.11. Weights must sum to exactly 1.0.

```python
def composite_vrp_score(signals, has_earnings_in_window=False, near_term_earnings=False):
    s = signals   # alias for readability

    # --- 12-signal weighted sum ---
    raw = (
        0.18 * normalize_signal(s.get('vrp_pctile', 0.0),      0.0,  1.0)   +  # Magnitude vs history
        0.12 * normalize_signal(s.get('vrp_persist_30d', 0.5), 0.4,  1.0)   +  # Reliability
        0.12 * normalize_signal(np.clip(s.get('vrp_zscore', 0.0), -3, 3),
                                -3.0, 3.0)                                    +  # Statistical significance
        0.10 * normalize_signal(s.get('em_ratio', 1.0),         0.8,  2.0)   +  # Implied vs realized move
        0.10 * normalize_signal(s.get('excess_vrp', 0.0),      -0.05, 0.10)  +  # Idiosyncratic premium
        0.08 * normalize_signal(s.get('ivp', 0.5),              0.0,  1.0)   +  # IV elevation
        0.08 * normalize_signal(s.get('skew_25d', 0.0),        -0.02, 0.08)  +  # Hedging demand
        0.07 * normalize_signal(s.get('term_slope_pctile', 0.5), 0.0, 1.0)  +  # Term structure support
        0.05 * normalize_signal(1.0 - s.get('jump_pct', 0.0),   0.5,  1.0)  +  # Jump cleanliness (inverted)
        0.04 * normalize_signal(s.get('pcr_oi', 1.0),           0.8,  2.5)  +  # Put buying demand
        0.03 * _gex_support_score(s.get('gex_billions', 0.0))               +  # Dealer positioning
        0.03 * normalize_signal(1.0 - s.get('vov_z', 0.0)/3.0,  0.0,  1.0)    # IV stability (inverted VoV)
    )
    # raw is in [0, 1]; scale to [0, 100]
    score = raw * 100.0

    # --- Multiplicative penalties (PRD §6.11) ---
    event_penalty = 0.30 if has_earnings_in_window else (0.15 if near_term_earnings else 0.0)
    jump_pct = s.get('jump_pct', 0.0)
    jump_penalty = 0.20 if jump_pct > 0.35 else (0.10 if jump_pct > 0.25 else 0.0)
    vov_z = s.get('vov_z', 0.0)
    vov_penalty = 1.0 if vov_z > 2.5 else (0.25 if vov_z > 1.5 else 0.0)

    if vov_z > 2.5:
        return 0.0   # DISQUALIFY — VoV too high

    score = score * (1.0 - event_penalty) * (1.0 - jump_penalty) * (1.0 - vov_penalty)
    return float(np.clip(score, 0.0, 100.0))

def _gex_support_score(gex_bn: float) -> float:
    """Convert GEX billions to [0,1] score for use in weighted sum."""
    if gex_bn > 0.5:   return 1.0
    if gex_bn > 0.0:   return 0.7
    if gex_bn > -0.5:  return 0.4
    return 0.0
```

```python
import numpy as np
import pandas as pd
import logging
from typing import Optional

logger = logging.getLogger(__name__)

def run_analytics(
    ticker: str,
    chain: dict,                     # from fetch_options_chain()
    ohlc: pd.DataFrame,              # from fetch_ohlcv()
    r: float,                        # risk-free rate from get_risk_free_rate()
    vix: float,                      # current VIX level
    iv30_history: Optional[pd.Series] = None,   # 252-day IV30 history (None = empty)
    vrp_history: Optional[pd.Series] = None,    # 252-day VRP history (None = empty)
    spy_vrp: float = 0.03,           # SPY VRP benchmark (default 3 vol pts)
    beta: float = 1.0,               # ticker beta (default 1.0)
    vvix: Optional[float] = None,    # VVIX for regime (optional)
    has_earnings_in_window: bool = False,
    near_term_earnings: bool = False,
) -> dict:
    """
    Full analytics pipeline for a single ticker. Returns a complete result dict
    or a dict with 'error' key if any required step fails.

    Call order (dependency chain):
    1. detect_regime(vix)
    2. compute_vrp(chain, ohlc, r)         [iv_surface.py]
    3. compute_vrp_signals(...)            [vrp_engine.py]
    4. composite_vrp_score(signals)        [composite_score.py]
    """
    result = {'ticker': ticker, 'error': None}

    # Step 1: Regime
    try:
        from analytics.regime import detect_regime
        regime = detect_regime(vix, vvix=vvix)
        result['regime'] = regime
    except Exception as e:
        logger.error(f"[{ticker}] regime detection failed: {e}")
        result['error'] = f'regime_failed: {e}'
        return result

    # Step 2: IV surface + VRP
    try:
        from analytics.iv_surface import compute_vrp
        vrp_data = compute_vrp(chain, ohlc, r)
        if vrp_data.get('error'):
            result['error'] = vrp_data['error']
            return result
        result.update({k: vrp_data[k] for k in ['iv30', 'iv60', 'iv30_td', 'vrp', 'ensemble_rv',
                                                  'har', 'garch', 'ewma', 'has_arbitrage']
                       if k in vrp_data})
    except Exception as e:
        logger.error(f"[{ticker}] IV surface failed: {e}")
        result['error'] = f'iv_surface_failed: {e}'
        return result

    # Step 3: All 12 VRP signals
    try:
        from analytics.vrp_engine import compute_vrp_signals, vrp_timing_signal
        _iv30_hist = iv30_history if iv30_history is not None else pd.Series([vrp_data['iv30']] * 30)
        _vrp_hist  = vrp_history  if vrp_history  is not None else pd.Series([vrp_data['vrp']]  * 30)

        signals = compute_vrp_signals(
            vrp_data=vrp_data, ohlc=ohlc, chain=chain,
            iv30_history=_iv30_hist, vrp_history=_vrp_hist,
            spy_vrp=spy_vrp, beta=beta, vix=vix, r=r,
        )
        result['signals'] = signals

        timing_action, timing_reason = vrp_timing_signal(
            signals.get('vrp_5d_change', 0.0),
            signals.get('vrp_10d_change', 0.0),
            signals.get('vrp_pctile', 0.5),
        )
        result['timing_action'] = timing_action
        result['timing_reason'] = timing_reason
    except Exception as e:
        logger.error(f"[{ticker}] VRP signals failed: {e}")
        result['error'] = f'vrp_signals_failed: {e}'
        return result

    # Step 4: Composite score
    try:
        from analytics.composite_score import composite_vrp_score
        score = composite_vrp_score(
            signals,
            has_earnings_in_window=has_earnings_in_window,
            near_term_earnings=near_term_earnings,
        )
        result['composite_score'] = score
    except Exception as e:
        logger.error(f"[{ticker}] composite score failed: {e}")
        result['error'] = f'composite_score_failed: {e}'
        return result

    # Convenience top-level keys for scanner (avoids nested dict access)
    result['vrp_pctile']     = signals.get('vrp_pctile', 0.0)
    result['ivp']            = signals.get('ivp', 0.5)
    result['gex_billions']   = signals.get('gex_billions', 0.0)
    result['pcr_oi']         = signals.get('pcr_oi', 1.0)
    result['jump_pct']       = signals.get('jump_pct', 0.0)
    result['regime_label']   = regime['label']
    result['regime_mult']    = regime['multiplier']

    return result
```

# Build synthetic chain
from scipy.stats import norm
spot, r = 450.0, 0.05
def make_exp(dte, base_iv):
    T = dte / 365.0
    strikes = np.arange(spot*0.88, spot*1.12, spot*0.025)
    rows = []
    for K in strikes:
        sigma = base_iv + 0.003*abs(K-spot)/spot
        d1 = (np.log(spot/K)+(r+sigma**2/2)*T)/(sigma*np.sqrt(T))
        d2 = d1 - sigma*np.sqrt(T)
        put = max(K*np.exp(-r*T)*norm.cdf(-d2) - spot*norm.cdf(-d1), 0.05)
        rows.append({'strike': K, 'bid': put*0.95, 'ask': put*1.05,
                     'openInterest': 500, 'volume': 200, 'gamma': 0.008,
                     'impliedVolatility': sigma})
    import pandas as pd_i
    df = pd_i.DataFrame(rows)
    return {'date': '2026-04-18', 'dte': dte, 'calls': df.copy(), 'puts': df}
