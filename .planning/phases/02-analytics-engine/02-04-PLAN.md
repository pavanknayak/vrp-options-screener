Exact five-regime logic from PRD §6.12:
```python
def detect_regime(vix, vvix=None, vix3m=None, vvix_history=None):
    # Compute VVIX Z-score if history available
    vvix_z = None
    if vvix is not None and vvix_history is not None and len(vvix_history) >= 30:
        vvix_z = (vvix - vvix_history.mean()) / vvix_history.std()

    # Primary VIX level thresholds (from PRD §6.12, exact values)
    if vix > 40:
        label, mult, msg = "Crisis",   0.00, "Close all. Do not open new short-vol."
    elif vvix_z is not None and vvix_z > 2.5:
        label, mult, msg = "VOL_UNSTABLE", 0.25, "VVIX spike: VRP unreliable. 25% size max."
    elif vix > 28:
        label, mult, msg = "High",     0.75, "Rich premium; tail risk elevated."
    elif vix > 20:
        label, mult, msg = "Elevated", 1.25, "Premium-rich: increase size modestly."
    elif vix > 15:
        label, mult, msg = "Normal",   1.00, "Standard environment."
    else:
        label, mult, msg = "Low",      0.50, "Thin premium; reduce size, be selective."

    # Term slope (VIX3M - VIX): contango = normal, backwardation = stress
    term_slope = (vix3m - vix) if vix3m is not None else None

    return {
        'label':      label,
        'multiplier': mult,
        'message':    msg,
        'vix':        vix,
        'vvix':       vvix,
        'vvix_z':     vvix_z,
        'term_slope': term_slope,
        'is_crisis':  label == "Crisis",
    }
```

The five valid labels are: "Low", "Normal", "Elevated", "High", "Crisis" (plus "VOL_UNSTABLE" as a special sub-regime). Callers check `result['label']` against these exact strings.

**vov_signal(iv30_history: pd.Series) -> dict**

Computes Vol-of-Vol from the rolling IV30 series (PRD §6.8):
```python
def vov_signal(iv30_history):
    if len(iv30_history) < 30:
        return {'vov_30d': None, 'vov_z': None, 'flag': 'insufficient_history'}

    vov_30d = float(iv30_history.iloc[-30:].std() * np.sqrt(252))  # annualized

    # Z-score vs. 252-day rolling VoV history
    rolling_vov = iv30_history.rolling(30).std() * np.sqrt(252)
    rolling_vov = rolling_vov.dropna()
    if len(rolling_vov) >= 21:
        vov_z = (vov_30d - float(rolling_vov.mean())) / max(float(rolling_vov.std()), 1e-8)
    else:
        vov_z = 0.0

    flag = ''
    if vov_z > 2.5:
        flag = 'NO-GO: VoV Z > 2.5 — IV extremely unstable'
    elif vov_z > 1.5:
        flag = 'Reduce position size 30% — VoV Z > 1.5'

    return {
        'vov_30d': vov_30d,
        'vov_z':   float(vov_z),
        'flag':    flag,
        'disqualify': vov_z > 2.5,
    }
```

**compute_gex(chain: dict, spot: float) -> dict**

Dealer Gamma Exposure from PRD §6.6. Uses the sign convention: dealers are assumed to be short calls (sold to retail) and short puts (sold to retail).

```python
def compute_gex(chain, spot):
    gex = 0.0
    for exp in chain['expirations']:
        # Dealers SHORT calls: their GEX contribution is NEGATIVE (short gamma)
        for _, row in exp['calls'].iterrows():
            oi    = row.get('openInterest', 0) or 0
            gamma = row.get('gamma', 0) or 0
            gex  += oi * gamma * 100 * spot**2 * (-1)   # dealer short call = -gamma

        # Dealers SHORT puts: their GEX contribution is POSITIVE (long gamma via delta hedge)
        for _, row in exp['puts'].iterrows():
            oi    = row.get('openInterest', 0) or 0
            gamma = row.get('gamma', 0) or 0
            gex  += oi * gamma * 100 * spot**2 * (+1)   # dealer short put = +gamma

    gex_bn = gex / 1e9

    # Interpret magnitude against PRD §6.6 thresholds
    if gex_bn > 0.5:
        interp = "Strongly supportive — dealer hedging suppresses vol"
    elif gex_bn > 0:
        interp = "Mildly supportive"
    elif gex_bn > -0.5:
        interp = "Mildly adverse — neutral on position size"
    else:
        interp = "Adverse — reduce position size 25%"

    return {
        'gex_raw':     gex,
        'gex_billions': gex_bn,
        'sign':        'positive' if gex > 0 else 'negative',
        'magnitude':   abs(gex_bn),
        'interpretation': interp,
        'supports_collection': gex > 0,
    }
```

**compute_pcr(chain: dict) -> dict**

Put-Call Ratio from PRD §6.7. Aggregate across all expirations in the chain:

```python
def compute_pcr(chain):
    total_put_oi  = 0
    total_call_oi = 0
    total_put_vol = 0
    total_call_vol = 0

    for exp in chain['expirations']:
        total_put_oi   += exp['puts']['openInterest'].sum()
        total_call_oi  += exp['calls']['openInterest'].sum()
        total_put_vol  += exp['puts'].get('volume', pd.Series([0])).sum()
        total_call_vol += exp['calls'].get('volume', pd.Series([0])).sum()

    pcr_oi  = total_put_oi  / max(total_call_oi,  1)
    pcr_vol = total_put_vol / max(total_call_vol, 1)

    # PRD §6.7 interpretation thresholds
    if pcr_oi > 1.8:
        interp = "Heavy institutional put buying — strong hedging demand premium"
    elif pcr_oi > 1.2:
        interp = "Elevated but normal — moderate hedging demand"
    elif pcr_oi >= 0.8:
        interp = "Balanced — structural premium present, less hedging-demand driven"
    else:
        interp = "Call-heavy — unusual; investigate for event rumors or M&A"

    return {
        'pcr_oi':        float(pcr_oi),
        'pcr_volume':    float(pcr_vol),
        'put_oi':        int(total_put_oi),
        'call_oi':       int(total_call_oi),
        'interpretation': interp,
    }
```
